#### Day 5: 기본 GraphRAG 통합

**목표**: Neo4j에서 직접 Cypher 쿼리 실행 기능 구현

**작업 내용**:

1. **GraphRAG 노드 구현**
   ```python
   from langchain_neo4j import Neo4jGraph
   
   class SimpleGraphRAG:
       def __init__(self, neo4j_uri, username, password):
           self.graph = Neo4jGraph(
               url=neo4j_uri,
               username=username,
               password=password
           )
       
       def query(self, natural_language_query):
           """자연어 질의를 Cypher로 변환하여 실행"""
           
           # 간단한 Cypher 생성 (나중에 LLM으로 개선)
           cypher = self._simple_cypher_generation(natural_language_query)
           
           # 쿼리 실행
           result = self.graph.query(cypher)
           
           return result
       
       def _simple_cypher_generation(self, query):
           """간단한 Cypher 생성 (기본 버전)"""
           # 나중에 LLM으로 개선
           if "기업" in query or "company" in query.lower():
               return "MATCH (c:Company) RETURN c LIMIT 10"
           return "MATCH (n) RETURN n LIMIT 10"
   ```

2. **LangGraph 노드로 통합**
   ```python
   def graphrag_query_node(state: ReportState) -> ReportState:
       """GraphRAG 쿼리 실행"""
       graphrag = SimpleGraphRAG(neo4j_uri, username, password)
       
       result = graphrag.query(state['query'])
       state['graphrag_results'] = result
       
       return state
   ```

#### Day 6-7: KG Construction Agent (간소화)

**목표**: 기본 엔티티 추출 및 Neo4j 주입 기능

**작업 내용**:

1. **기본 엔티티 추출**
   ```python
   from langchain_openai import ChatOpenAI
   
   class KGConstructionAgent:
       def __init__(self, llm, neo4j_client):
           self.llm = llm
           self.neo4j = neo4j_client
       
       def extract_entities(self, document):
           """문서에서 엔티티 추출"""
           
           prompt = f"""
           다음 문서에서 엔티티를 추출하세요.
           
           문서:
           {document}
           
           추출할 엔티티:
           - Company (기업)
           - ProductLine (제품)
           - Metric (재무 지표)
           
           JSON 형식:
           {{
               "companies": [{{"name": "...", "properties": {{}}}}],
               "products": [{{"name": "...", "properties": {{}}}}],
               "metrics": [{{"name": "...", "value": "...", "properties": {{}}}}]
           }}
           """
           
           response = self.llm.invoke(prompt)
           entities = json.loads(response.content)
           
           return entities
       
       def inject_to_neo4j(self, entities):
           """추출된 엔티티를 Neo4j에 주입"""
           
           for company in entities.get('companies', []):
               self.neo4j.run("""
                   MERGE (c:Company {name: $name})
                   SET c += $properties
               """, name=company['name'], properties=company.get('properties', {}))
           
           # 제품, 지표도 동일하게 처리
   ```

2. **LangGraph 노드로 변환**
   ```python
   def kg_construction_node(state: ReportState) -> ReportState:
       """KG Construction Agent 실행"""
       agent = KGConstructionAgent(llm, neo4j_client)
       
       # 문서에서 엔티티 추출
       entities = agent.extract_entities(state.get('document', ''))
       
       # Neo4j에 주입
       agent.inject_to_neo4j(entities)
       
       state['kg_data'] = entities
       return state
   ```

   ```python
   class SectorAnalystAgent:
       def __init__(self, llm, graphrag):
           self.llm = llm
           self.graphrag = graphrag
       
       def analyze(self, query, graphrag_results):
           """섹터 분석 수행"""
           
           # GraphRAG 결과를 텍스트로 변환
           context = self._format_graphrag_results(graphrag_results)
           
           prompt = f"""
           당신은 반도체 섹터 전문 분석가입니다.
           
           질의: {query}
           
           관련 정보:
           {context}
           
           다음을 포함한 섹터 리포트를 작성하세요:
           1. 섹터 전체 동향
           2. 주요 기업 비교
           3. 밸류체인 분석
           4. 리스크 요인
           5. 투자 포인트
           """
           
           report = self.llm.invoke(prompt)
           return report.content
       
       def _format_graphrag_results(self, results):
           """GraphRAG 결과 포맷팅"""
           # 그래프 데이터를 읽기 쉬운 텍스트로 변환
           text = "Companies:\n"
           for company in results.get('companies', []):
               text += f"- {company['name']}: {company.get('industry', '')}\n"
           return text
   ```

#### Day 10: Ontology Architect Agent + Quality Check

**목표**: 온톨로지 스키마 자동 생성 및 품질 검사

**작업 내용**:

1. **Ontology Architect Agent**
   ```python
   from neo4j_graphrag import SchemaFromTextExtractor
   
   class OntologyArchitectAgent:
       def __init__(self, llm, neo4j_client):
           self.llm = llm
           self.neo4j = neo4j_client
           self.extractor = SchemaFromTextExtractor(llm=llm)
       
       def generate_schema(self, templates):
           """스키마 자동 생성"""
           
           # neo4j-graphrag 활용
           schema = self.extractor.extract_schema(documents=templates)
           
           # Neo4j 스키마로 변환
           neo4j_schema = schema.to_neo4j_schema()
           
           # Neo4j에 적용
           for statement in neo4j_schema.to_cypher():
               self.neo4j.run(statement)
           
           return schema
   ```

2. **Quality Check Agent**
   ```python
   class QualityCheckAgent:
       def __init__(self, llm, neo4j_client):
           self.llm = llm
           self.neo4j = neo4j_client
       
       def check(self, graph_data):
           """그래프 품질 검사"""
           
           issues = []
           
           # 1. 스키마 일관성 검사
           schema_issues = self._check_schema_consistency()
           issues.extend(schema_issues)
           
           # 2. 엔티티 중복 검사
           duplicate_issues = self._check_duplicates()
           issues.extend(duplicate_issues)
           
           # 3. 관계 완전성 검사
           completeness_issues = self._check_completeness()
           issues.extend(completeness_issues)
           
           return {
               'issues': issues,
               'has_issues': len(issues) > 0
           }
       
       def _check_schema_consistency(self):
           """스키마 일관성 검사"""
           # Neo4j에서 스키마 정보 조회
           result = self.neo4j.run("""
               CALL db.schema.visualization()
               YIELD nodes, relationships
               RETURN nodes, relationships
           """)
           
           # 스키마 이슈 감지 로직
           issues = []
           # ... 검사 로직
           
           return issues
   ```

3. **조건부 라우팅 구현**
   ```python
   def route_after_quality_check(state: ReportState) -> str:
       """품질 검사 후 라우팅"""
       
       quality_result = state.get('quality_check_result', {})
       
       if quality_result.get('has_issues'):
           # 스키마 이슈가 있으면 Ontology Architect로
           schema_issues = [
               issue for issue in quality_result.get('issues', [])
               if 'schema' in issue.get('type', '').lower()

           if schema_issues:
               return "ontology_architect"
       
       # 문제 없으면 GraphRAG로
       return "graphrag_query"
   
   # 조건부 엣지 추가
   workflow.add_conditional_edges(
       "quality_check",
       route_after_quality_check,
       {
           "ontology_architect": "ontology_architect",
           "graphrag_query": "graphrag_query"
       }
   )
   ```

#### Day 13: 리포트 포맷팅 및 출력

**목표**: 마크다운 리포트 생성 및 저장

**작업 내용**:

1. **리포트 생성 노드**
   ```python
   def report_generation_node(state: ReportState) -> ReportState:
       """최종 리포트 생성"""
       
       report_parts = []
       
       # 섹터 분석 추가
       if state.get('sector_analysis'):
           report_parts.append(f"# 섹터 분석\n\n{state['sector_analysis']}")
       
       # 종목별 분석 추가
       if state.get('company_analysis'):
           report_parts.append("# 종목별 분석\n\n")
           for company, analysis in state['company_analysis'].items():
               report_parts.append(f"## {company}\n\n{analysis}\n\n")
       
       # 리포트 결합
       final_report = "\n\n".join(report_parts)
       
       # 메타데이터 추가
       metadata = f"""
       ---
       생성일: {datetime.now().isoformat()}
       리포트 타입: {state.get('report_type', 'sector')}
       대상 기업: {', '.join(state.get('target_companies', []))}
       ---
       
       """
       
       state['final_report'] = metadata + final_report
       
       # 파일로 저장
       save_report(state['final_report'], state.get('report_type', 'sector'))
       
       return state
   
   def save_report(report, report_type):
       """리포트를 파일로 저장"""
       timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
   ```

```python
# scripts/setup_neo4j.py
from neo4j import GraphDatabase
import os
from dotenv import load_dotenv

load_dotenv()

def setup_neo4j():
    """Neo4j 초기 설정"""
    
    driver = GraphDatabase.driver(
        os.getenv("NEO4J_URI"),
        auth=(os.getenv("NEO4J_USER"), os.getenv("NEO4J_PASSWORD"))
    )
    
    with driver.session() as session:
        # 제약 조건 생성
        session.run("""
            CREATE CONSTRAINT company_name IF NOT EXISTS
            FOR (c:Company) REQUIRE c.name IS UNIQUE;
        """)
        
        # 인덱스 생성
        session.run("""
            CREATE INDEX company_industry IF NOT EXISTS
            FOR (c:Company) ON (c.industry);
        """)
    
    print("✅ Neo4j 설정 완료")
    driver.close()

if __name__ == "__main__":
    setup_neo4j()
```

### 3. 샘플 데이터 로드

```python
# scripts/load_sample_data.py
import json
from neo4j import GraphDatabase

def load_sample_data():
    """샘플 데이터 로드"""
    
    # 샘플 데이터 읽기
    with open("data/raw/sample_companies.json", "r", encoding="utf-8") as f:
        companies = json.load(f)
    
    driver = GraphDatabase.driver(
        os.getenv("NEO4J_URI"),
        auth=(os.getenv("NEO4J_USER"), os.getenv("NEO4J_PASSWORD"))
    )
    
    with driver.session() as session:
        for company in companies:
            # Company 노드 생성
            session.run("""
                MERGE (c:Company {name: $name})
                SET c.industry = $industry,
                    c.exchange = $exchange,
                    c.marketCap = $marketCap
            """, **company)
            
            # 제품 관계 생성
            for product in company.get('products', []):
                session.run("""
                    MERGE (p:ProductLine {name: $product_name})
                    MERGE (c:Company {name: $company_name})
                    MERGE (c)-[:MANUFACTURES]->(p)
                """, product_name=product, company_name=company['name'])
    
    print("✅ 샘플 데이터 로드 완료")
    driver.close()
```
